    let audioCtx = null;

    // Khởi tạo và resume AudioContext ngay trong cử chỉ của người dùng
    function ensureAudioContext() {
      try {
        if (!audioCtx) {
          const AudioContextClass = window.AudioContext || window.webkitAudioContext;
          if (!AudioContextClass) {
            updateAudioStatus('Trình duyệt không hỗ trợ Web Audio API', false);
            return null;
          }
          audioCtx = new AudioContextClass();
        }
        if (audioCtx.state === 'suspended') {
          audioCtx.resume().then(() => {
            updateAudioStatus('🔊 Âm thanh đã bật sẵn sàng!', true);
          }).catch(() => {});
        } else if (audioCtx.state === 'running') {
          updateAudioStatus('🔊 Âm thanh đang hoạt động', true);
        }
        return audioCtx;
      } catch (err) {
        updateAudioStatus('Không thể khởi tạo âm thanh lúc này', false);
        return null;
      }
    }

            gain.gain.exponentialRampToValueAtTime(0.28, now + idx * 0.11 + 0.02);
            gain.gain.exponentialRampToValueAtTime(0.001, now + idx * 0.11 + 0.24);
            osc.connect(gain);
            gain.connect(ctx.destination);
            osc.start(now + idx * 0.11);
            osc.stop(now + idx * 0.11 + 0.25);
          });
          updateAudioStatus('🔊 Đã phát tiếng thử thành công!', true);
        } catch (e) {}
      };

      if (ctx.state === 'suspended') {
        ctx.resume().then(doPlayTest).catch(doPlayTest);
      } else {
        doPlayTest();
      }
    }

    function playTone(type) {
      if (!soundEnabled) return;
      try {
        const ctx = ensureAudioContext();
        if (!ctx) return;

        const trigger = () => {
          const now = ctx.currentTime;
          const osc = ctx.createOscillator();
          const gain = ctx.createGain();
          osc.connect(gain);
          gain.connect(ctx.destination);

          if (type === 'click') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(580, now);
            osc.frequency.exponentialRampToValueAtTime(820, now + 0.07);
            gain.gain.setValueAtTime(0.2, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.08);
            osc.start(now);
            osc.stop(now + 0.09);
          } else if (type === 'whittle') {
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(280, now);
            osc.frequency.exponentialRampToValueAtTime(560, now + 0.14);
            gain.gain.setValueAtTime(0.25, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.16);
            osc.start(now);
            osc.stop(now + 0.17);
          } else if (type === 'dye') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(360, now);
            osc.frequency.exponentialRampToValueAtTime(720, now + 0.2);
            gain.gain.setValueAtTime(0.25, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.22);
            osc.start(now);
            osc.stop(now + 0.23);
          } else if (type === 'snip') {
            osc.type = 'sawtooth';
            osc.frequency.setValueAtTime(880, now);
            osc.frequency.exponentialRampToValueAtTime(340, now + 0.09);
            gain.gain.setValueAtTime(0.18, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.1);
            osc.start(now);
            osc.stop(now + 0.11);
          } else if (type === 'press') {
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(220, now);
            osc.frequency.exponentialRampToValueAtTime(110, now + 0.22);
            gain.gain.setValueAtTime(0.28, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.24);
            osc.start(now);
            osc.stop(now + 0.25);
          } else if (type === 'place' || type === 'chime') {
            osc.type = 'sine';
            osc.frequency.setValueAtTime(523.25, now);
            osc.frequency.setValueAtTime(659.25, now + 0.09);
            osc.frequency.setValueAtTime(783.99, now + 0.18);
            gain.gain.setValueAtTime(0.24, now);
            gain.gain.exponentialRampToValueAtTime(0.001, now + 0.34);
            osc.start(now);
            osc.stop(now + 0.35);
          } else if (type === 'win') {
            const notes = [523.25, 659.25, 783.99, 1046.5, 783.99, 1046.5];
            notes.forEach((freq, idx) => {
              const o = ctx.createOscillator();
              const g = ctx.createGain();
              o.type = 'triangle';
              o.connect(g);
              g.connect(ctx.destination);
              const t0 = now + idx * 0.13;
              o.frequency.setValueAtTime(freq, t0);
              g.gain.setValueAtTime(0.001, t0);
              g.gain.exponentialRampToValueAtTime(0.26, t0 + 0.02);
              g.gain.exponentialRampToValueAtTime(0.001, t0 + 0.36);
              o.start(t0);
              o.stop(t0 + 0.38);
            });
          }
        };

        if (ctx.state === 'suspended') {
          ctx.resume().then(trigger).catch(trigger);
        } else {
          trigger();
        }
      } catch (e) {}
    }

      for (let i = 0; i < count; i++) {
        const el = document.createElement('div');
        el.className = 'particle text-lg sm:text-xl';
        el.textContent = emojis[Math.floor(Math.random() * emojis.length)];
        el.style.left = x + 'px';
        el.style.top = y + 'px';
        el.style.setProperty('--dx', (Math.random() - 0.5) * 160 + 'px');
        el.style.setProperty('--dy', (Math.random() * 110 + 30) + 'px');
        el.style.setProperty('--rot', (Math.random() - 0.5) * 360 + 'deg');
        document.body.appendChild(el);
        setTimeout(() => el.remove(), 950);
      }
    }

          document.getElementById('splinter-left').setAttribute('opacity', '1');
          finished.setAttribute('opacity', '0.5');

          document.getElementById('splinter-right').setAttribute('opacity', '1');
          finished.setAttribute('opacity', '0.8');

          nextBtn.disabled = false;
          nextBtn.classList.add('animate-hint');
        }
      }

      zone.addEventListener('pointerdown', (e) => {
        isDragging = true;
        doWhittle(e.clientX, e.clientY);
      });

      zone.addEventListener('pointermove', (e) => {
        if (!isDragging) return;
        const rect = zone.getBoundingClientRect();

        knife.style.right = 'auto';
        knife.style.bottom = 'auto';

        if (x > rect.width * 0.22 && x < rect.width * 0.65) {
          doWhittle(e.clientX, e.clientY);
        }

    function renderTreeTray() {
      const tray = document.getElementById('tree-tray');
      tray.innerHTML = '';
      treeBloomConfigs.forEach((cfg, idx) => {
        const placed = state.treePlaced[idx];
        const btn = document.createElement('button');
        btn.disabled = placed;
        btn.onclick = () => placeTreeBloom(idx);
        btn.className = `toy-btn bg-white p-1.5 rounded-2xl flex flex-col items-center justify-center ${placed ? 'opacity-30' : 'hover:scale-105'}`;
        btn.innerHTML = `
          <svg viewBox="-28 -28 56 56" class="w-9 h-9">
            ${buildMiniBloomSVG({ ...cfg, radius: 19 })}
          </svg>
          <span class="text-[10px] font-bold text-[#5C3824] mt-0.5">${cfg.label}</span>
        `;
        tray.appendChild(btn);
      });
    }

    function placeTreeBloom(idx) {
      if (state.treePlaced[idx]) {
        const nextFree = state.treePlaced.findIndex(v => !v);
        if (nextFree === -1) return;
        idx = nextFree;
      }
      state.treePlaced[idx] = true;
      const slot = document.getElementById('tree-slot-' + idx);
      slot.innerHTML = buildMiniBloomSVG(treeBloomConfigs[idx]);
      playTone('place');

      const rect = slot.getBoundingClientRect();
      spawnParticles(rect.left + rect.width / 2, rect.top + rect.height / 2, ['🌺', '🌼', '✨'], 5);

      const doneCount = state.treePlaced.filter(Boolean).length;
      document.getElementById('tree-bloom-count').textContent = `${doneCount} / 8`;
      renderTreeTray();

      if (doneCount >= 8) {
        document.getElementById('quick-tree-btn').disabled = true;
        const nextBtn = document.getElementById('next-step-5');
        nextBtn.disabled = false;
        nextBtn.classList.add('animate-hint');
      }
    }

    function autoPlaceTreeBloom() {
      const idx = state.treePlaced.findIndex(v => !v);
      if (idx !== -1) placeTreeBloom(idx);
    }
